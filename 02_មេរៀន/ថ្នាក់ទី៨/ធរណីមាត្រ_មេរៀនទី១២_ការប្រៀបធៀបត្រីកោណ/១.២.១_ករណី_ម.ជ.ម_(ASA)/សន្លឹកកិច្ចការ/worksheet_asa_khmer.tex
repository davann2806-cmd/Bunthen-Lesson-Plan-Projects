\newcommand{\precutcard}{%
  \begin{tikzpicture}[scale=1.0]
    % ប្រអប់បន្ទាត់ដាច់ៗសម្រាប់កាត់ (Dashed border around card)
    \draw[dashed, line width=0.65pt, gray!70, rounded corners=2pt] (-0.65, -0.9) rectangle (4.65, 3.15);

    % ស្លាកកន្ត្រៃកាត់
    \node[anchor=north east, font=\fontsize{6.5pt}{8pt}\selectfont, text=gray!80] at (4.55, 3.08) {\ding{36} កាត់តាមបន្ទាត់នេះ};

    % កូអរដោនេកំពូលទាំង ៣ នៃត្រីកោណ A'B'C'
    \coordinate (Ap) at (0, 0);
    \coordinate (Bp) at (4, 0);
    \coordinate (Cp) at (2.175, 2.175);

    % ផ្ទៃពណ៌ស្រាលខាងក្នុង និងគំនូសជ្រុងទាំង ៣
    \fill[teal!8] (Ap) -- (Bp) -- (Cp) -- cycle;
    \draw[line width=1.3pt, teal!80!black, line join=round] (Ap) -- (Bp) -- (Cp) -- cycle;

    % ធ្នូមុំ 45 ដឺក្រេ ត្រង់ A'
    \draw[line width=0.6pt, teal!70!black] (0.55, 0) arc (0:45:0.55);
    \node[font=\fontsize{7pt}{8pt}\selectfont, text=teal!90!black] at (0.85, 0.28) {$45^\circ$};

    % ធ្នូមុំ 50 ដឺក្រេ ត្រង់ B'
    \draw[line width=0.6pt, teal!70!black] (3.45, 0) arc (180:130:0.55);
    \node[font=\fontsize{7pt}{8pt}\selectfont, text=teal!90!black] at (3.15, 0.28) {$50^\circ$};

    % ចំណុចកំពូល
    \fill (Ap) circle (1.8pt);
    \fill (Bp) circle (1.8pt);
    \fill (Cp) circle (1.8pt);

    % ឈ្មោះកំពូល
    \node[below left=1pt, font=\bfseries\small] at (Ap) {$A'$};
    \node[below right=1pt, font=\bfseries\small] at (Bp) {$B'$};
    \node[above=2pt, font=\bfseries\small] at (Cp) {$C'$};
    
    % ប្រវែងបាត 4 cm
    \node[below=2.5pt, font=\fontsize{7.5pt}{9pt}\selectfont] at (2, 0) {$4\text{ cm}$};
  \end{tikzpicture}%
}
