\newcommand{\precutcard}{%
  \begin{tikzpicture}[scale=1.0]
    % ប្រអប់បន្ទាត់ដាច់ៗសម្រាប់កាត់ (Dashed border around card)
    \draw[dashed, line width=0.65pt, gray!70, rounded corners=2pt] (-0.65, -0.9) rectangle (4.65, 6.15);

    % ស្លាកកន្ត្រៃកាត់ (ដាក់ខាងលើឆ្វេង ដើម្បីកុំឱ្យជាន់ឈ្មោះ C')
    \node[anchor=north west, font=\fontsize{6.5pt}{8pt}\selectfont, text=gray!80] at (-0.55, 6.08) {\ding{36} កាត់តាមបន្ទាត់នេះ};

    % កូអរដោនេកំពូលទាំង ៣ នៃត្រីកោណ A'B'C'
    \coordinate (Ap) at (0, 0);
    \coordinate (Bp) at (4, 0);
    \coordinate (Cp) at (3, 5.196);

    % ផ្ទៃពណ៌ស្រាលខាងក្នុង និងគំនូសជ្រុងទាំង ៣
    \fill[teal!8] (Ap) -- (Bp) -- (Cp) -- cycle;
    \draw[line width=1.3pt, teal!80!black, line join=round] (Ap) -- (Bp) -- (Cp) -- cycle;

    % ធ្នូមុំ 60 ដឺក្រេ ត្រង់ A'
    \draw[line width=0.6pt, teal!70!black] (0.8, 0) arc (0:60:0.8);
    \node[font=\fontsize{7pt}{8pt}\selectfont, text=teal!90!black] at (1.2, 0.5) {$60^\circ$};

    % ចំណុចកំពូល
    \fill (Ap) circle (1.8pt);
    \fill (Bp) circle (1.8pt);
    \fill (Cp) circle (1.8pt);

    % ឈ្មោះកំពូល
    \node[below left=1pt, font=\bfseries\small] at (Ap) {$A'$};
    \node[below right=1pt, font=\bfseries\small] at (Bp) {$B'$};
    \node[above=2pt, font=\bfseries\small] at (Cp) {$C'$};

    % ប្រវែងជ្រុង A'B' = 4 cm និង A'C' = 6 cm
    \node[below=2.5pt, font=\fontsize{7.5pt}{9pt}\selectfont] at (2, 0) {$4\text{ cm}$};
    \node[rotate=60, font=\fontsize{7.5pt}{9pt}\selectfont] at ($(Ap)!0.5!(Cp) + (150:0.28)$) {$6\text{ cm}$};
  \end{tikzpicture}%
}
